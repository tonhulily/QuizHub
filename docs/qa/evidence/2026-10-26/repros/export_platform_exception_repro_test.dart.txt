const _exam = ExamDetail(
  examId: 9,
  title: 'Repro export',
  questionCount: 1,
  questions: [
    ExamQuestion(
      order: 1,
      questionId: 3,
      content: 'Question',
      options: [QuestionOption(key: 'A', text: 'Answer')],
    ),
  ],
);

class _ExportRepo implements ExamRepository {
  bool exported = false;

  @override
  Future<ExportResult> exportFile(
    int examId, {
    ExportFormat format = ExportFormat.pdf,
    bool includeAnswerKey = false,
    String paperSize = 'A4',
  }) async {
    exported = true;
    return const ExportResult(
      downloadUrl: 'https://files.example.test/exam.pdf',
    );
  }

  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}

class _ThrowingLauncher extends Fake
    with MockPlatformInterfaceMixin
    implements UrlLauncherPlatform {
  @override
  Future<bool> launchUrl(String url, LaunchOptions options) async =>
      throw PlatformException(
        code: 'ACTIVITY_NOT_FOUND',
        message: 'No Activity found to handle intent',
      );
}

void main() {
  testWidgets('export launch PlatformException escapes without app feedback', (
    tester,
  ) async {
    final originalLauncher = UrlLauncherPlatform.instance;
    tester.view.physicalSize = const Size(800, 1200);
    tester.view.devicePixelRatio = 1;
    final repo = _ExportRepo();
    getIt.registerSingleton<ExamRepository>(repo);
    UrlLauncherPlatform.instance = _ThrowingLauncher();
    addTearDown(() async {
      UrlLauncherPlatform.instance = originalLauncher;
      tester.view.resetPhysicalSize();
      tester.view.resetDevicePixelRatio();
      await getIt.reset();
    });

    await tester.pumpWidget(
      MaterialApp(
        theme: AppTheme.lightTheme,
        builder: EasyLoading.init(),
        home: withRole(
          UserRole.lecturer,
          const ExamDetailView(examId: 9, initial: _exam),
        ),
      ),
    );
    await tester.pumpAndSettle();
    await tester.tap(find.text('Xuất PDF / Excel'));
    await tester.pumpAndSettle();
    await tester.tap(find.text('Xuất file'));
    await tester.pumpAndSettle();
  });
}
